Lab 6: Using Ansible Templates
Objectives
By the end of this lab, you will be able to:

• Understand the fundamentals of Jinja2 templating in Ansible • Create dynamic configuration files using Jinja2 templates • Use Ansible facts and variables within templates • Deploy templates to target hosts using the template module • Implement conditional logic and loops in templates • Troubleshoot common template-related issues

Prerequisites
Before starting this lab, you should have:

• Basic understanding of Ansible playbooks and modules • Familiarity with YAML syntax • Knowledge of Linux file system and basic commands • Understanding of web server configuration concepts • Completion of previous Ansible labs or equivalent experience

Lab Environment
Ready-to-Use Cloud Machines: Al Nafi provides pre-configured Linux-based cloud machines for this lab. Simply click Start Lab to access your environment - no need to build your own VM or install software.

Your lab environment includes: • Control Node: CentOS/RHEL 8 with Ansible pre-installed • Managed Nodes: 2 CentOS/RHEL 8 servers for testing • All necessary packages and dependencies pre-installed

Subtask 1.4: Create the Jinja2 Template for httpd.conf
Create a comprehensive Apache configuration template:
cat > templates/httpd.conf.j2 << 'EOF'
#
# Apache HTTP Server Configuration File
# Generated by Ansible Template on {{ ansible_date_time.iso8601 }}
# Managed by: {{ ansible_user }}@{{ ansible_hostname }}
#

# Basic server configuration
ServerRoot "/etc/httpd"
Listen {{ apache_listen_port }}
{% if enable_ssl %}
Listen {{ ssl_port }} ssl
{% endif %}

# Dynamic module loading based on host variables
{% if custom_modules is defined %}
{% for module in custom_modules %}
LoadModule {{ module }}_module modules/{{ module }}.so
{% endfor %}
{% endif %}

# User and Group
User apache
Group apache

# Server identification
ServerAdmin {{ apache_server_admin }}
ServerName {{ apache_server_name }}:{{ apache_port }}

# Document root and directory permissions
DocumentRoot "{{ apache_document_root }}"

<Directory "{{ apache_document_root }}">
    Options Indexes FollowSymLinks
    AllowOverride None
    Require all granted
</Directory>

# Performance tuning based on environment
{% if app_environment == 'production' %}
# Production settings
MaxRequestWorkers {{ apache_max_clients }}
ThreadsPerChild {{ apache_threads_per_child }}
Timeout {{ apache_timeout }}
KeepAlive On
MaxKeepAliveRequests 100
KeepAliveTimeout 15
{% else %}
# Development settings
MaxRequestWorkers {{ apache_max_clients }}
ThreadsPerChild 10
Timeout 60
KeepAlive Off
{% endif %}

# Logging configuration
ErrorLog "logs/error_log"
LogLevel {{ apache_log_level }}

# Custom log format with system information
LogFormat "%h %l %u %t \"%r\" %>s %O \"%{Referer}i\" \"%{User-Agent}i\" [{{ ansible_hostname }}]" combined
CustomLog "logs/access_log" combined

# Server signature with dynamic information
ServerTokens Prod
ServerSignature Off

# Application-specific configuration
<Directory "{{ apache_document_root }}/{{ app_name | lower | replace(' ', '_') }}">
    Options -Indexes +FollowSymLinks
    AllowOverride All
    Require all granted
    
    # Environment-specific headers
    {% if app_environment == 'production' %}
    Header always set X-Environment "Production"
    Header always set X-App-Version "{{ app_version }}"
    {% else %}
    Header always set X-Environment "Development"
    Header always set X-Debug "Enabled"
    {% endif %}
</Directory>

# System information comments (for debugging)
# System: {{ ansible_distribution }} {{ ansible_distribution_version }}
# Architecture: {{ ansible_architecture }}
# Processor: {{ ansible_processor[1] if ansible_processor|length > 1 else 'Unknown' }}
# Memory: {{ ansible_memtotal_mb }}MB
# Generated on: {{ ansible_date_time.date }} at {{ ansible_date_time.time }}

# Include additional configuration files
IncludeOptional conf.d/*.conf
EOF

Task 2: Apply the Template using ansible-playbook
Subtask 2.1: Create the Main Playbook
Create a comprehensive playbook to deploy Apache with templates:
cat > deploy-apache-templates.yml << 'EOF'
---
- name: Deploy Apache Web Server with Jinja2 Templates
  hosts: webservers
  become: yes
  gather_facts: yes
  
  tasks:
    - name: Display system information
      debug:
        msg: |
          Deploying to: {{ ansible_hostname }}
          OS: {{ ansible_distribution }} {{ ansible_distribution_version }}
          IP: {{ ansible_default_ipv4.address }}
          Environment: {{ app_environment }}

    - name: Install Apache HTTP Server
      yum:
        name: httpd
        state: present
      notify: restart apache

    - name: Install additional packages
      yum:
        name:
          - httpd-tools
          - mod_ssl
        state: present

    - name: Create backup of original httpd.conf
      copy:
        src: /etc/httpd/conf/httpd.conf
        dest: /etc/httpd/conf/httpd.conf.backup
        remote_src: yes
        backup: yes
      ignore_errors: yes

    - name: Deploy Apache configuration from template
      template:
        src: httpd.conf.j2
        dest: /etc/httpd/conf/httpd.conf
        owner: root
        group: root
        mode: '0644'
        backup: yes
      notify: restart apache

    - name: Create application directory
      file:
        path: "{{ apache_document_root }}/{{ app_name | lower | replace(' ', '_') }}"
        state: directory
        owner: apache
        group: apache
        mode: '0755'

    - name: Deploy index.html from template
      template:
        src: index.html.j2
        dest: "{{ apache_document_root }}/index.html"
        owner: apache
        group: apache
        mode: '0644'

    - name: Deploy application-specific index.html
      template:
        src: index.html.j2
        dest: "{{ apache_document_root }}/{{ app_name | lower | replace(' ', '_') }}/index.html"
        owner: apache
        group: apache
        mode: '0644'

    - name: Ensure Apache is started and enabled
      systemd:
        name: httpd
        state: started
        enabled: yes

    - name: Open firewall for HTTP
      firewalld:
        service: http
        permanent: yes
        state: enabled
        immediate: yes
      ignore_errors: yes

    - name: Open firewall for HTTPS (if SSL enabled)
      firewalld:
        service: https
        permanent: yes
        state: enabled
        immediate: yes
      when: enable_ssl
      ignore_errors: yes

    - name: Validate Apache configuration
      command: httpd -t
      register: apache_config_test
      changed_when: false

    - name: Display configuration test results
      debug:
        var: apache_config_test.stdout_lines

  handlers:
    - name: restart apache
      systemd:
        name: httpd
        state: restarted
EOF
Subtask 2.2: Run the Playbook
Execute the playbook with verbose output:
ansible-playbook -i inventory deploy-apache-templates.yml -v
Check the playbook execution results and verify no errors occurred.

Subtask 3.3: Create a Template with Conditional Logic
Create an advanced Apache virtual host template:
cat > templates/vhost.conf.j2 << 'EOF'
# Virtual Host Configuration for {{ ansible_hostname }}
# Generated on {{ ansible_date_time.iso8601 }}

{% set server_name = ansible_fqdn %}
{% set doc_root = apache_document_root %}

<VirtualHost *:{{ apache_listen_port }}>
    ServerName {{ server_name }}
    {% if ansible_hostname != ansible_fqdn %}
    ServerAlias {{ ansible_hostname }}
    {% endif %}
    
    DocumentRoot {{ doc_root }}
    
    # Environment-specific configuration
    {% if app_environment == 'production' %}
    # Production settings
    LogLevel error
    ServerSignature Off
    
    # Security headers
    Header always set X-Content-Type-Options nosniff
    Header always set X-Frame-Options DENY
    Header always set X-XSS-Protection "1; mode=block"
    
    {% elif app_environment == 'development' %}
    # Development settings
    LogLevel debug
    ServerSignature On
    
    # Development headers
    Header always set X-Debug-Mode "Enabled"
    Header always set X-Developer "Ansible-Generated"
    
    {% else %}
    # Default settings
    LogLevel warn
    ServerSignature Off
    {% endif %}
    
    # Custom error pages based on system resources
    {% if ansible_memtotal_mb > 1000 %}
    ErrorDocument 404 /errors/404.html
    ErrorDocument 500 /errors/500.html
    {% else %}
    ErrorDocument 404 "Resource not found on {{ ansible_hostname }}"
    ErrorDocument 500 "Server error on {{ ansible_hostname }}"
    {% endif %}
    
    # Directory configuration with dynamic permissions
    <Directory "{{ doc_root }}">
        Options {% if app_environment == 'development' %}Indexes {% endif %}FollowSymLinks
        AllowOverride {% if app_environment == 'production' %}None{% else %}All{% endif %}
        Require all granted
        
        {% if custom_modules is defined and 'mod_rewrite' in custom_modules %}
        # Rewrite rules (mod_rewrite available)
        RewriteEngine On
        RewriteCond %{REQUEST_FILENAME} !-f
        RewriteCond %{REQUEST_FILENAME} !-d
        RewriteRule ^(.*)$ /index.html [L]
        {% endif %}
    </Directory>
    
    # Logging with hostname
    ErrorLog logs/{{ ansible_hostname }}_error.log
    CustomLog logs/{{ ansible_hostname }}_access.log combined
    
    # Performance tuning based on system specs
    {% if ansible_processor_vcpus >= 4 %}
    # Multi-core system optimizations
    ThreadsPerChild {{ apache_threads_per_child * 2 }}
    {% else %}
    # Single/dual-core system settings
    ThreadsPerChild {{ apache_threads_per_child }}
    {% endif %}
    
</VirtualHost>

{% if enable_ssl %}
# SSL Virtual Host
<VirtualHost *:{{ ssl_port }}>
    ServerName {{ server_name }}
    DocumentRoot {{ doc_root }}
    
    SSLEngine on
    SSLCertificateFile /etc/pki/tls/certs/localhost.crt
    SSLCertificateKeyFile /etc/pki/tls/private/localhost.key
    
    # SSL Security based on environment
    {% if app_environment == 'production' %}
    SSLProtocol all -SSLv2 -SSLv3 -TLSv1 -TLSv1.1
    SSLCipherSuite ECDHE-ECDSA-AES256-GCM-SHA384:ECDHE-RSA-AES256-GCM-SHA384
    {% else %}
    SSLProtocol all -SSLv2 -SSLv3
    {% endif %}
    
    ErrorLog logs/{{ ansible_hostname }}_ssl_error.log
    CustomLog logs/{{ ansible_hostname }}_ssl_access.log combined
</VirtualHost>
{% endif %}
EOF
Create a playbook to deploy the virtual host:
cat > deploy-vhost.yml << 'EOF'
---
- name: Deploy Apache Virtual Host Configuration
  hosts: webservers
  become: yes
  gather_facts: yes

    - name: Display configuration test results

    - name: reload apache

        name: httpd
        state: reloaded
EOF
